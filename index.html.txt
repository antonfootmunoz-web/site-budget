    <script>
        const searchInput = document.getElementById('search-input');
        const priceInput = document.getElementById('price-input');
        const priceValue = document.getElementById('price-value');
        const resultsContainer = document.getElementById('results-container');

        function handleSearch() {
            const query = searchInput.value.trim();
            const budgetMax = parseInt(priceInput.value);
            
            // Met à jour l'affichage du prix en direct
            priceValue.textContent = budgetMax + " €";

            if (query === "") {
                resultsContainer.innerHTML = '<div class="welcome-msg">En attente de votre recherche... Tape un mot pour générer l\'objet !</div>';
                return;
            }

            // Calcul du prix automatique basé sur le mot tapé
            let hash = 0;
            for (let i = 0; i < query.length; i++) {
                hash = query.charCodeAt(i) + ((hash << 5) - hash);
            }
            let generatedPrice = Math.abs(hash % 1400) + 15; 

            // Adapte le prix si ça dépasse le budget
            if (generatedPrice > budgetMax) {
                generatedPrice = Math.round(budgetMax * 0.9);
            }

            // On injecte le résultat sans bug
            resultsContainer.innerHTML = `
                <article class="product-card">
                    <div class="badge-container">
                        <span class="price-tag">` + generatedPrice + ` €</span>
                        <span class="stock-tag">✔️ Meilleur prix trouvé</span>
                    </div>
                    <h3>📦 ` + query + `</h3>
                    <p>Voici l'offre exclusive trouvée pour "` + query + `". Cet article respecte parfaitement ton budget maximum de ` + budgetMax + ` €.</p>
                    <a href="https://www.amazon.fr/s?k=` + encodeURIComponent(query) + `" target="_blank" class="buy-btn">🛒 Voir et acheter sur Amazon</a>
                </article>
            `;
        }

        searchInput.addEventListener('input', handleSearch);
        priceInput.addEventListener('input', handleSearch);
    </script>
